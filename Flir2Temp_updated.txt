import cv2
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import os
import matplotlib.patches as patches
import tkinter as tk
from tkinter import simpledialog
import math

# Configuration for temperature range and ROI sizes
CONFIG = {
    "min_temperature": 17,
    "max_temperature": 42,
    "roi_shapes": {
        "nose": "circle",
        "forehead": "box"
    },
    "roi_sizes": {
        "nose": (10, 10),
        "forehead": (50, 20)
    }
}

# Initialize global variables
data = pd.DataFrame(columns=['X', 'Y', 'ROI Type', 'Temperature'])
circles = []
rectangles = []
temperature_annotations = []
selected_roi_type = "nose"
interactive_elements = {
    "current_circle": None,
    "current_forehead_box": None
}


def get_temperature(pixel_value, min_value, max_value):
    """Convert pixel value to temperature."""
    return min_value + (pixel_value / 255) * (max_value - min_value)


def calculate_average_temperature(x, y, gray_image):
    """Calculate the average temperature for the selected ROI."""
    global selected_roi_type

    roi_type = selected_roi_type
    roi_shape = CONFIG["roi_shapes"][roi_type]
    roi_size = CONFIG["roi_sizes"][roi_type]

    if roi_shape == "circle":
        radius = roi_size[0] // 2
        y_indices, x_indices = np.ogrid[:gray_image.shape[0], :gray_image.shape[1]]
        distances = np.sqrt((x_indices - x) ** 2 + (y_indices - y) ** 2)
        mask = distances <= radius
        area = gray_image[mask]
    else:
        width, height = roi_size
        x_start = max(0, x - width // 2)
        y_start = max(0, y - height // 2)
        x_end = min(gray_image.shape[1], x + width // 2)
        y_end = min(gray_image.shape[0], y + height // 2)
        area = gray_image[y_start:y_end, x_start:x_end]

    temperatures = [get_temperature(pixel_value, CONFIG["min_temperature"], CONFIG["max_temperature"]) 
                    for pixel_value in area.flatten()]
    return np.mean(temperatures)


def update_temperature_display(x, y, roi_type, temperature):
    """Display the temperature as an annotation on the plot."""
    truncated_temperature = round(temperature, 1)
    return plt.gca().annotate(f"{roi_type}: {truncated_temperature}°C", xy=(x, y), xytext=(x + 30, y - 30),
                              color='white', ha='left', va='bottom',
                              arrowprops=dict(arrowstyle='-', color='red', alpha=0.7),
                              bbox=dict(facecolor='red', alpha=0.7))


def handle_roi_selection():
    """Prompt user for ROI type and validate input."""
    root = tk.Tk()
    root.withdraw()
    roi_type = simpledialog.askstring("ROI Type", "Enter the ROI type (nose or forehead):")
    root.destroy()
    if roi_type and roi_type.lower() in CONFIG["roi_sizes"]:
        return roi_type.lower()
    else:
        print("Invalid ROI type. Defaulting to 'nose'.")
        return "nose"


def on_click(event, gray_image):
    """Handle mouse click event to select ROI and annotate temperature."""
    global selected_roi_type, data

    if event.button == 1 and event.inaxes is not None:
        x, y = int(event.xdata), int(event.ydata)
        selected_roi_type = handle_roi_selection()
        avg_temp = calculate_average_temperature(x, y, gray_image)

        if CONFIG["roi_shapes"][selected_roi_type] == "circle":
            radius = CONFIG["roi_sizes"][selected_roi_type][0] // 2
            circle = patches.Circle((x, y), radius, edgecolor='red', linewidth=1, facecolor='none')
            plt.gca().add_patch(circle)
            circles.append(circle)
        else:
            width, height = CONFIG["roi_sizes"][selected_roi_type]
            rect = patches.Rectangle((x - width // 2, y - height // 2), width, height, edgecolor='red', linewidth=1, facecolor='none')
            plt.gca().add_patch(rect)
            rectangles.append(rect)

        annotation = update_temperature_display(x, y, selected_roi_type, avg_temp)
        temperature_annotations.append(annotation)

        data.loc[len(data)] = [x, y, selected_roi_type, avg_temp]
        plt.draw()


def on_move(event):
    """Handle mouse movement to update ROI preview."""
    global interactive_elements

    if event.inaxes is not None:
        x, y = int(event.xdata), int(event.ydata)

        # Update dynamic circle (for "nose")
        if interactive_elements["current_circle"]:
            interactive_elements["current_circle"].center = (x, y)

        # Update dynamic rectangle (for "forehead")
        width, height = CONFIG["roi_sizes"]["forehead"]
        if interactive_elements["current_forehead_box"]:
            interactive_elements["current_forehead_box"].set_xy((x - width // 2, y - height // 2))

        plt.draw()


def on_scroll(event):
    """Handle mouse scroll to zoom in and out of the image."""
    if event.inaxes is not None and event.inaxes.images:
        xlim, ylim = plt.gca().get_xlim(), plt.gca().get_ylim()
        scale_factor = 1.3 if event.button == 'down' else 0.8
        x, y = event.xdata, event.ydata
        plt.gca().set_xlim((xlim[0] - x) * scale_factor + x, (xlim[1] - x) * scale_factor + x)
        plt.gca().set_ylim((ylim[0] - y) * scale_factor + y, (ylim[1] - y) * scale_factor + y)
        plt.draw()


def initialize_interactive_elements():
    """Initialize elements like dynamic ROI indicators."""
    global interactive_elements

    nose_radius = CONFIG["roi_sizes"]["nose"][0] // 2
    interactive_elements["current_circle"] = patches.Circle((0, 0), nose_radius, edgecolor='orange', facecolor='none')

    forehead_width, forehead_height = CONFIG["roi_sizes"]["forehead"]
    interactive_elements["current_forehead_box"] = patches.Rectangle((0, 0), forehead_width, forehead_height, edgecolor='green', facecolor='none')

    plt.gca().add_patch(interactive_elements["current_circle"])
    plt.gca().add_patch(interactive_elements["current_forehead_box"])


def process_image(image_path):
    """Main function to process the image and extract temperatures."""
    global data

    if not os.path.exists(image_path):
        raise FileNotFoundError(f"Image path not found: {image_path}")

    image_name = os.path.basename(image_path)
    gray_image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    if gray_image is None:
        raise ValueError("Failed to load image. Ensure it is a valid grayscale image.")

    smooth_image = cv2.GaussianBlur(gray_image, (5, 5), 0)
    plt.imshow(smooth_image, cmap='gray')
    plt.axis('off')

    initialize_interactive_elements()

    plt.gcf().canvas.mpl_connect('button_press_event', lambda event: on_click(event, gray_image))
    plt.gcf().canvas.mpl_connect('motion_notify_event', on_move)
    plt.gcf().canvas.mpl_connect('scroll_event', on_scroll)

    plt.show()
    data.to_excel(f'temperature_data_{image_name}.xlsx', index=False)
    print("Data saved successfully.")
